<script>
    async function fetchPublications() {
        const response = await fetch('https://scholar.google.com/citations?user=dAyogGsAAAAJ&hl=en&view_op=list_works&sortby=pubdate');
        const text = await response.text();
        const parser = new DOMParser();
        const doc = parser.parseFromString(text, 'text/html');
        const publications = doc.querySelectorAll('.gsc_a_tr');
        let publicationList = '';

        publications.forEach(pub => {
            const title = pub.querySelector('.gsc_a_at').textContent;
            const link = pub.querySelector('.gsc_a_at').href;
            const journal = pub.querySelector('.gs_gray').textContent;
            const year = pub.querySelector('.gsc_a_h').textContent;
            publicationList += `<li><strong>${title}</strong> - <em>${journal}</em>, ${year}. <a href="${link}">Link</a></li>`;
        });

        document.getElementById('publications').innerHTML = publicationList;
    }

    document.addEventListener('DOMContentLoaded', fetchPublications);
</script>
